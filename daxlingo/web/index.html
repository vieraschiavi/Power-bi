» ventas vs año anterior
VAR Actual = SUM ( Ventas[Importe] )
VAR Anterior =
    CALCULATE (
        SUM ( Ventas[Importe] ),
        SAMEPERIODLASTYEAR ( Calendario[Fecha] )
    )
RETURN
    DIVIDE ( Actual - Anterior, Anterior )
